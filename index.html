Google Sheets — Live Sync
Auto-push every entry to Google Sheets
Connect a Google Sheet once — every new expense entry will be appended automatically. Works with your personal Gmail, no OAuth required.
One-time setup (5 minutes):
① Go to sheets.google.com → create a new sheet called Riwayat
② In the sheet, click Extensions → Apps Script
③ Paste the script below → click Deploy → New deployment → Web app
④ Set "Who has access" to Anyone → copy the Web App URL
⑤ Paste the URL in the field below → Save
function doPost(e){ const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const tabName = data.sheet || 'Entries'; let sheet = ss.getSheetByName(tabName); if(!sheet){ sheet = ss.insertSheet(tabName); } const headers = { 'Entries': ['Date','Supplier','Txn Type','Event', 'Category','Item','Invoice Amt','Paid Amt', 'Pay Mode','Paid By','Remark'], 'Bills': ['Date','Supplier','Event','Bill Amt', 'Invoice Ref','Description','Entered By'], 'Payments': ['Date','Supplier','Event','Amount', 'Type','Pay Mode','Paid By','Bill Ref','Remark'], 'Returns': ['Date','Supplier','Items Returned','Amount', 'Status','Event','Bill Ref','Remark','Entered By'] }; if(sheet.getLastRow()===0 && headers[tabName]){ sheet.appendRow(headers[tabName]); sheet.getRange(1,1,1,headers[tabName].length) .setFontWeight('bold') .setBackground('#1a3a1a') .setFontColor('#ffffff'); } const rows = { 'Entries': [data.date,data.supplier,data.txn_type, data.event,data.category,data.item, data.invoice_amt,data.paid_amt, data.pay_mode,data.paid_by,data.remark], 'Bills': [data.date,data.supplier,data.event, data.amount,data.bill_ref,data.description, data.entered_by], 'Payments': [data.date,data.supplier,data.event, data.amount,data.type,data.mode, data.paid_by,data.bill_ref,data.remark], 'Returns': [data.date,data.supplier,data.items, data.amount,data.status,data.event, data.bill_ref,data.remark,data.entered_by] }; sheet.appendRow(rows[tabName] || Object.values(data)); return ContentService .createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); }
